function convertUrl(url, type) { // Trata links do YouTube if (url.includes('youtube.com') || url.includes('youtu.be')) { let videoId = ""; if (url.includes('youtu.be/')) { videoId = url.split('youtu.be/')[1].split('?')[0]; } else if (url.includes('v=')) { videoId = url.split('v=')[1].split('&')[0]; } return `https://www.youtube.com/embed/${videoId}?autoplay=1&mute=1&controls=0&loop=1&playlist=${videoId}`; } // Trata links do Google Drive const driveMatch = url.match(/\/file\/d\/([a-zA-Z0-9_-]+)|id=([a-zA-Z0-9_-]+)/); if (driveMatch) { const fileId = driveMatch[1] || driveMatch[2]; if (type === 'video') { // Link direto de streaming do arquivo no Drive return `https://drive.google.com/uc?export=download&id=${fileId}`; } else { return `https://lh3.googleusercontent.com/d/${fileId}`; } } return url; }
